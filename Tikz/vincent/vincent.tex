    \coordinate (B4) at ($(A4) + \scaleA*(p)$);
    \coordinate (C4) at ($(B4) - \H*\gamma*(x) + 2*\H*\delta/\cx*(x)$);
    \coordinate (D4) at ($(C4) - \scaleA*(p)$);

    \coordinate (F4) at ($(E4) - \H*\gamma*(x) + 2*\H*\delta/\cx*(x)$);
